UnityEditor.EditorApplication.delayCall += () => {
try {
var report=new System.Text.StringBuilder();
var sheets=new[]{"Classics","Events","Newcomers"};
var groups=new[]{
new[]{"billiards","hockey","bowling","airhockey","wallball","cornhole","cheese","dock","keepaway","curling","base","terrier","minefield","mattress","wife","parking"},
new[]{"swing","buzkashi","bog","bungee","fishing","sumo","target","fieldgoal","pong","poker","tug","skee","log","dodge","derby","plowing"},
new[]{"cabchaos","kickball","danceoff","farefight","skytow","limbo","trailerplug","yourcrew"}};
for(int sheet=0;sheet<sheets.Length;sheet++){
 string path="Assets/TruckSports/Art/SportIcons/"+sheets[sheet]+".png";
 var importer=(UnityEditor.TextureImporter)UnityEditor.AssetImporter.GetAtPath(path);
 importer.textureType=UnityEditor.TextureImporterType.Sprite;importer.spriteImportMode=UnityEditor.SpriteImportMode.Multiple;
 importer.alphaIsTransparency=true;importer.mipmapEnabled=false;importer.npotScale=UnityEditor.TextureImporterNPOTScale.None;importer.maxTextureSize=2048;importer.textureCompression=UnityEditor.TextureImporterCompression.Uncompressed;importer.wrapMode=UnityEngine.TextureWrapMode.Clamp;
 importer.SaveAndReimport();
 var factory=new UnityEditor.U2D.Sprites.SpriteDataProviderFactories();factory.Init();var provider=factory.GetSpriteEditorDataProviderFromObject(importer);
 if(provider==null)throw new System.InvalidOperationException("Sprite provider unavailable; no slicing attempted: "+path);
 provider.InitSpriteEditorDataProvider();var editing=provider.GetDataProvider<UnityEditor.U2D.Sprites.ISpriteFrameEditCapability>();
 if(editing==null)throw new System.InvalidOperationException("Sprite edit capability unavailable: "+path);
 var cap=editing.GetEditCapability();
 if(!cap.HasCapability(UnityEditor.U2D.Sprites.EEditCapability.CreateAndDeleteSprite)||!cap.HasCapability(UnityEditor.U2D.Sprites.EEditCapability.EditSpriteName)||!cap.HasCapability(UnityEditor.U2D.Sprites.EEditCapability.EditSpriteRect)||!cap.HasCapability(UnityEditor.U2D.Sprites.EEditCapability.EditPivot))throw new System.InvalidOperationException("Native sprite capability denied. No metadata changes applied: "+path);
 importer.GetSourceTextureWidthAndHeight(out int width,out int height);int rows=groups[sheet].Length/4;
 var old=provider.GetSpriteRects();var frames=new UnityEditor.SpriteRect[groups[sheet].Length];
 var pairs=new System.Collections.Generic.List<UnityEditor.SpriteNameFileIdPair>();
 for(int i=0;i<frames.Length;i++){string name="sport-"+groups[sheet][i];var id=UnityEngine.GUID.Generate();foreach(var previous in old)if(previous.name==name)id=previous.spriteID;
  int x0=i%4*width/4,x1=(i%4+1)*width/4,y0=height-(i/4+1)*height/rows,y1=height-i/4*height/rows;
  frames[i]=new UnityEditor.SpriteRect{name=name,spriteID=id,rect=new UnityEngine.Rect(x0,y0,x1-x0,y1-y0),alignment=(int)UnityEngine.SpriteAlignment.Center,pivot=new UnityEngine.Vector2(.5f,.5f)};
  pairs.Add(new UnityEditor.SpriteNameFileIdPair(name,id));
 }
 var names=provider.GetDataProvider<UnityEditor.U2D.Sprites.ISpriteNameFileIdDataProvider>();if(names==null)throw new System.InvalidOperationException("Stable sprite name/file IDs unavailable; operation aborted.");
 provider.SetSpriteRects(frames);names.SetNameFileIdPairs(pairs);provider.Apply();importer.SaveAndReimport();
 int count=0;foreach(var asset in UnityEditor.AssetDatabase.LoadAllAssetsAtPath(path))if(asset is UnityEngine.Sprite)count++;
 if(count!=frames.Length)throw new System.InvalidOperationException("Imported sprite count mismatch: "+path);
 report.AppendLine("PASS "+path+" sprites="+count+" source="+width+"x"+height);
}
System.IO.File.WriteAllText("Logs/TruckSports/EssentialRelease/Sprite-slicing.txt",report.ToString());
}catch(System.Exception e){System.IO.File.WriteAllText("Logs/TruckSports/EssentialRelease/Sprite-slicing.txt","FAIL "+e);UnityEngine.Debug.LogException(e);}
};
