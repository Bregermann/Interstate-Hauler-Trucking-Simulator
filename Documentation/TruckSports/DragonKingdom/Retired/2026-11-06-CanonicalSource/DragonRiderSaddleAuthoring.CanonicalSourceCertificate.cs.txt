        static string CertifiedCanonicalSourceRowExpected(CanonicalMountSourceAuthority authority,string path,string recordedSha256,bool testOnly)
        {
            if(authority==null)return recordedSha256;
            // No metadata, prefab, controller or runtime path receives a
            // substitute hash. Their existing exact comparisons remain sole
            // authority. Test tokens bind their own disposable path only.
            if(path.Replace('\\','/')!=authority.proof.target)return recordedSha256;

        }

        {public string scope="TEST_ONLY_REUSES_ACTUAL_V426_SAVED_RESULT_NO_CANONICAL_WRITES_OR_NEW_EIGHT_REPLACEMENTS";public bool savedCopyValidated,publicReaderRejectsDisposable,productionGuardRejectsTestAuthority,defaultChangedRowRejected,otherRowsStrict,sourceBytesExact;public int headerFaultsRejected;public CanonicalSourceCertificateProof savedCopy;}
        public static string ProbeCanonicalMountSourceCertificateControls(string evidence)
        {
            RequireRiderPassedManifest(evidence,"DragonRiderCanonicalMountPromotionTests.DisposableCanonicalPlanApplyPreservesOriginalsAndRejectsOrRollsBackFaults(\"baseline\")");string path=Path.Combine(evidence,"rendered-evidence","RiderSaddleAuthoring","CanonicalMountPromotion","mount-promotion-baseline.json");var before=File.ReadAllBytes(path);var original=File.ReadAllBytes(DragonRidersAuthoring.ScenePath);var meta=File.ReadAllBytes(DragonRidersAuthoring.ScenePath+".meta");var apply=JsonUtility.FromJson<CanonicalMountApply>(File.ReadAllText(path));var output=new CanonicalSourceControlProof();
            try
            {
                try{ValidateCanonicalMountPromotionSourceCertificate(path);}catch(InvalidOperationException){output.publicReaderRejectsDisposable=true;}if(!output.publicReaderRejectsDisposable)throw new InvalidOperationException("Disposable source must never create public canonical authority.");
                var authority=ReadCanonicalMountSourceAuthority(path,apply,true,evidence);output.savedCopy=authority.proof;output.savedCopyValidated=authority.proof.currentProfilesExact&&authority.proof.sourceClosureExact&&!authority.proof.productionEligible;
                try{RequireProductionCanonicalSourceAuthority(authority);}catch(InvalidOperationException){output.productionGuardRejectsTestAuthority=true;}
                string Expected(CanonicalMountSourceAuthority token,string source,string recorded)=>CertifiedCanonicalSourceRowExpected(token,source,recorded,true);if(!CanonicalCertificateSame(Expected(authority,apply.target,apply.sourceSha256),apply.resultSha256))throw new InvalidOperationException("Saved copy old/new row matcher failed.");
                output.defaultChangedRowRejected=!CanonicalCertificateSame(Hash(apply.target),Expected(null,apply.target,apply.sourceSha256));output.otherRowsStrict=Expected(authority,apply.target+".meta",apply.plan.metaSha256)==apply.plan.metaSha256&&Expected(authority,DragonRidersAuthoring.ScenePath,apply.sourceSha256)==apply.sourceSha256&&Expected(authority,"Assets/Unrelated.prefab","unchanged")=="unchanged";
                try{Expected(authority,apply.target,"WRONG_ORIGINAL_SHA");}catch(InvalidOperationException){output.headerFaultsRejected++;}
                foreach(string fault in new[]{"old-sha","new-sha","metadata","project","target","source-closure","profiles","world","config"})
                {
                    var changed=JsonUtility.FromJson<CanonicalMountApply>(File.ReadAllText(path));if(fault=="old-sha")changed.sourceSha256="WRONG";if(fault=="new-sha")changed.resultSha256="WRONG";if(fault=="metadata")changed.metaSha256="WRONG";if(fault=="project")changed.projectRoot="F:/WrongProject";if(fault=="target")changed.target="Assets/Unrelated.unity";if(fault=="source-closure")changed.sourceClosureExact=false;if(fault=="profiles")changed.profilesExact=false;if(fault=="world")changed.worldExact=false;if(fault=="config")changed.configurationExact=false;try{RequireCanonicalCertificateHeader(changed,true);}catch(InvalidOperationException){output.headerFaultsRejected++;}
                }
                if(!output.savedCopyValidated||!output.productionGuardRejectsTestAuthority||!output.defaultChangedRowRejected||!output.otherRowsStrict||output.headerFaultsRejected!=10)throw new InvalidOperationException("Canonical single-row source controls failed.");
            }
