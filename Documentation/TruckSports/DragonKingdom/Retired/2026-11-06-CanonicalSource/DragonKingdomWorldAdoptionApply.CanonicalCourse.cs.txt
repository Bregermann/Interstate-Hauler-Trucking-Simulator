        static void ValidateCanonicalMountedCourseEvidence(string tag,string path,DragonKingdomMountedCourseReview.Receipt proof)
        {
            string promotion=Arg("-tsDragonCanonicalMountPromotionReceipt",""),tests=Arg("-tsDragonMountedCourseTests","");

            string folder=Root+"VisualCandidates/"+tag;RequireCanonicalMountedCourseMetadata(proof,tag,promotion,live,Digest("Assets/TruckDragonRiders/Scenes/TruckDragonRiders.unity"),Digest(folder+"/Shared Dragon Kingdom.prefab"),Digest(folder+"/DragonKingdomPreview.unity"),Digest(folder+"/Integration Receipt.json"));
