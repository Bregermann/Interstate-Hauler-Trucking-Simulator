    static void Validate(object proof,object live)=>Plan.GetMethod("RequireCanonicalMountedCourseMetadata",BindingFlags.Static|BindingFlags.NonPublic).Invoke(null,new[]{proof,"MetadataOnly",Promotion,JsonUtility.ToJson(live,true),Hash,Hash,Hash,Hash});
    [Test]
    public void WellFormedRecordedMetadataKeepsFreshCountersZeroButCannotByItselfAuthorizeAdoption()
    {
        var proof=Metadata(out var live);Validate(proof,live);foreach(var rig in (Array)Get(proof,"rigs"))foreach(string field in new[]{"poses","baseFlightPoses","configuredGraphPoses","nativeMixerPoses"})Assert.That(Get(rig,field),Is.EqualTo(0));
        var error=Assert.Throws<TargetInvocationException>(()=>Plan.GetMethod("ValidateCanonicalMountedCourseEvidence",BindingFlags.Static|BindingFlags.NonPublic).Invoke(null,new[]{"MetadataOnly","Logs/nonexistent-native-receipt.json",proof}));Assert.That(error.InnerException,Is.Not.Null,"Metadata cannot bypass required native XML/live source certificate.");
    }
    [TestCase("disposable")][TestCase("world")][TestCase("scene")][TestCase("receipt")][TestCase("rig-count")][TestCase("duplicate-crew")][TestCase("fresh-counter")][TestCase("poses")][TestCase("mixer")][TestCase("stale-source")][TestCase("source-duplicate")][TestCase("wrong-sha")][TestCase("restore")][TestCase("controller")][TestCase("clip")][TestCase("course")][TestCase("profile")][TestCase("unknown-world-certificate")]
    public void RecordedCanonicalMetadataRejectsIndependentAuthorityAndProvenanceMutation(string fault)
    {
        var proof=Metadata(out var live);var q=Get(proof,"qualifiedSource");var rig=((Array)Get(proof,"rigs")).GetValue(0);
        switch(fault)
        {
            case "disposable":Set(live,"productionEligible",false);break;
            case "world":Set(proof,"worldPrefabSha256",new string('b',64));break;
            case "scene":Set(proof,"ridersScene","Assets/Wrong.unity");break;
            case "receipt":Set(proof,"canonicalPromotionReceipt","Logs/Wrong.json");break;
            case "rig-count":Set(proof,"rigs",Array.CreateInstance(rig.GetType(),7));break;
            case "duplicate-crew":Set(((Array)Get(proof,"rigs")).GetValue(2),"crew",0);break;
            case "fresh-counter":Set(rig,"poses",699);break;
            case "poses":Set(rig,"recordedSourcePoses",698);break;
            case "mixer":Set(rig,"recordedMixerLabelledPoses",0);break;
            case "stale-source":Set(q,"currentSourceBytesExact",false);break;
            case "source-duplicate":var files=(Array)Get(q,"currentSourceSnapshot");files.SetValue(files.GetValue(0),1);break;
            case "wrong-sha":Set(proof,"ridersSceneSha256","INVALID");break;
            case "restore":Set(q,"comparisonClosedBeforeCandidateReopen",false);break;
            case "controller":Set(rig,"controllerSha256",new string('b',64));break;
            case "clip":Set(rig,"clips",new[]{"Logs/other-clip"});break;
            case "course":Set(proof,"originalCourseHash","CHANGED");break;
            case "profile":Set(q,"currentProfilesExact",false);break;
            case "unknown-world-certificate":Set(live,"currentWorldExact",false);break;
        }
        var error=Assert.Throws<TargetInvocationException>(()=>Validate(proof,live));Assert.That(error.InnerException,Is.TypeOf<InvalidOperationException>());
    }
    [Test]
    public void CanonicalSnapshotUsesOnePhysicalKeyForRelativeAbsoluteAliasesWhileHistoricalPathsRemainUnchanged()
    {
        var method=Author.GetMethod("QualifiedCourseSnapshotPaths",BindingFlags.NonPublic|BindingFlags.Static);var paths=new[]{Canonical,Path.GetFullPath(Canonical)};
        var canonical=(string[])method.Invoke(null,new object[]{paths,true});var historical=(string[])method.Invoke(null,new object[]{paths,false});Assert.That(canonical.Length,Is.EqualTo(1));Assert.That(canonical[0],Is.EqualTo(Path.GetFullPath(Canonical)));CollectionAssert.AreEqual(paths,historical,"Default historical path representation remains unchanged.");
    }
    [Test]
